rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Usuários: lê e escreve só o próprio dado
    match /usuarios/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // Agendamentos: usuário lê/escreve os próprios, admin lê/escreve todos
    match /agendamentos/{id} {
      allow read, write: if request.auth != null && (
        resource == null ||
        resource.data.uid == request.auth.uid ||
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.isAdmin == true
      );
      allow create: if request.auth != null;
    }

    // Clientes, produtos e itens de fiado: só admin
    match /clientes/{clienteId} {
      allow read, write: if request.auth != null &&
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.isAdmin == true;

      match /itens/{itemId} {
        allow read, write: if request.auth != null &&
          get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.isAdmin == true;
      }
    }

    match /produtos/{id} {
      allow read, write: if request.auth != null &&
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.isAdmin == true;
    }
  }
}
